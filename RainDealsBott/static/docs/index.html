Authentication Header

Authorization: Bearer <API_KEY>
Routing Notice: Do not use /api or /api/v1 alone as root requests since they return 404 Not Found. Always append full sub-paths (e.g., /api/v1/products).

Important Manual Delivery Protocol

Customer Action Required Before Ordering
  • For manual delivery orders, the recipient must first open @raindeliverybot on Telegram and press Start.
  • Why? Telegram protocol strictly prevents bots from sending unsolicited direct messages or keys to a user who has not initiated a conversation first.
  • The API request must pass delivery_telegram_id with the recipient's exact numeric Telegram ID.
  • Security Rule: Account logins, codes, or credentials for manual items are delivered exclusively through Telegram and are never exposed in the API JSON response.

Expected Status & Error Responses

HTTP Code Status Description & Cause Sample JSON Response
200 OK Request completed successfully. {"success": true, ...}
400 Bad Request Invalid params, invalid quantity, missing fields. {"detail": "Invalid quantity or missing parameter"}
401 Unauthorized Missing or malformed Authorization header. {"detail": "Not authenticated"}
403 Forbidden API key is invalid, revoked, or inactive. {"detail": "Invalid or inactive API key"}
404 Not Found Order ID or endpoint path does not exist. {"detail": "Order not found"}
409 Conflict Duplicate non-idempotent action or state mismatch. {"detail": "Insufficient balance or conflicting state"}
422 Unprocessable Entity Validation fail (e.g., string passed instead of int). {"detail": [{"loc": ["body", "quantity"], "msg": "value is not integer"}]}
429 Too Many Requests Rate limit exceeded. Try again later. {"detail": "Rate limit exceeded"}
503 Service Unavailable Database or backend service temporarily offline. {"detail": "Service temporarily unavailable"}

import time
import uuid
import requests

BASE_URL = "https://raindealsbotapi.onrender.com"
API_KEY = "PASTE_YOUR_API_KEY_HERE"

def test_create_auto_order(service_id, quantity):
    url = f"{BASE_URL}/api/v1/order"
    headers = {
        "Authorization": f"Bearer {API_KEY}",
        "Content-Type": "application/json",
        "Accept": "application/json"

    payload = {
        "service_id": service_id,
        "quantity": quantity,
        "client_order_id": client_order_id
    }

    print("Sending order request...")
    response = requests.post(url, headers=headers, json=payload, timeout=30)
    print("Status Code:", response.status_code)
    print("Response JSON:", response.json())

if __name__ == "__main__":

Safe Retry Logic (Idempotency)

Zero Double-Charging

Demonstration of safely retrying order creation using identical client_order_id to prevent double charges during network drops.

test_idempotency_retry.py 
import requests

BASE_URL = "https://raindealsbotapi.onrender.com"
API_KEY = "PASTE_YOUR_API_KEY_HERE"

def test_safe_retry():
    url = f"{BASE_URL}/api/v1/order"
    headers = {
        "Authorization": f"Bearer {API_KEY}",
        "Content-Type": "application/json",
        "Accept": "application/json"
    }

    # Static client_order_id for idempotency test
    fixed_client_order_id = "test-tx-retry-unique-id-99881122"

    payload = {
        "service_id": 1,
        "quantity": 1,
        "client_order_id": fixed_client_order_id
    }

    print("--- 1. Initial Attempt ---")
    res1 = requests.post(url, headers=headers, json=payload, timeout=30)
    print("Initial Status:", res1.status_code)
    print("Initial Response:", res1.json())

    print("\n--- 2. Network Timeout Retry Attempt (Same client_order_id) ---")
    res2 = requests.post(url, headers=headers, json=payload, timeout=30)
    print("Retry Status:", res2.status_code)
    print("Retry Response (Returns existing order without charging twice):", res2.json())

if __name__ == "__main__":
    test_safe_retry()

raindeals_full_api.py 
# raindeals_full_api.py
# Install once: pip install requests

import time
import uuid
import requests


BASE_URL = "https://raindealsbotapi.onrender.com"
API_KEY = "PASTE_YOUR_API_KEY_HERE"


class RainDealsAPI:
    def __init__(self):
        self.headers = {
            "Authorization": f"Bearer {API_KEY}",
            "Content-Type": "application/json",
            "Accept": "application/json",
        }

    def request(self, method, path, payload=None, use_api_key=True):
        headers = self.headers if use_api_key else {
            "Accept": "application/json",
        }

        try:
            response = requests.request(
                method=method,
                url=f"{BASE_URL}{path}",
                headers=headers,
                json=payload,
                timeout=30,
            )
        except requests.RequestException as error:
            print("\nConnection error:", error)
            return None, None

        print(f"\n{method} {path}")
        print("Status:", response.status_code)

        try:
            data = response.json()
            print(data)
        except ValueError:
            data = None
            print(response.text)

        return response.status_code, data

    def health(self):
        return self.request("GET", "/health", use_api_key=False)

    def root(self):
        return self.request("GET", "/", use_api_key=False)

    def get_me(self):
        return self.request("GET", "/api/v1/me")

    def get_products(self):
        return self.request("GET", "/api/v1/products")

    def get_orders(self, page, limit):
        return self.request(
            "GET",
            f"/api/v1/orders?page={page}&limit={limit}",
        )

    def get_order(self, order_id):
        return self.request(
            "GET",
            f"/api/v1/order/{order_id}",
        )

    def create_order(
        self,
        service_id,
        quantity,
        delivery_telegram_id=None,
    ):
        client_order_id = (
            f"order-{int(time.time())}-{uuid.uuid4().hex[:12]}"
        )

        payload = {
            "service_id": service_id,
            "quantity": quantity,
            "client_order_id": client_order_id,
        }

        if delivery_telegram_id:
            payload["delivery_telegram_id"] = delivery_telegram_id

        return self.request(
            "POST",
            "/api/v1/order",
            payload=payload,
        )


def get_number(message, minimum=1, maximum=None):
    while True:
        try:
            value = int(input(message).strip())

            if value < minimum:
                print(f"Enter a number greater than or equal to {minimum}.")
                continue

            if maximum and value > maximum:
                print(f"Enter a number less than or equal to {maximum}.")
                continue

            return value

        except ValueError:
            print("Enter a valid number.")


def confirm_real_order():
    print("\nWARNING: This creates a REAL paid order.")
    return input("Type ORDER to continue: ").strip() == "ORDER"


def show_menu():
    print("\n" + "=" * 42)
    print("RainDeals API Test Menu")
    print("=" * 42)
    print("1. Check API health")
    print("2. Check API root")
    print("3. Get account and wallet balance")
    print("4. Get all products")
    print("5. Get order history")
    print("6. Get one order")
    print("7. Create automatic delivery order")
    print("8. Create manual delivery order")
    print("0. Exit")


def main():
    if API_KEY == "PASTE_YOUR_API_KEY_HERE":
        print("Paste your API key into API_KEY first.")
        return

    api = RainDealsAPI()

    while True:
        show_menu()
        choice = input("\nChoose an option: ").strip()

        if choice == "1":
            api.health()

        elif choice == "2":
            api.root()

        elif choice == "3":
            api.get_me()

        elif choice == "4":
            status, result = api.get_products()

            if status == 200 and result:
                for product in result.get("services", []):
                    print(
                        f"\nID: {product['service_id']}"
                        f"\nName: {product['name']}"
                        f"\nPrice: {product['price']} {product['currency']}"
                        f"\nStock: {product['stock']}"
                        f"\nDelivery: {product['delivery_type']}"
                    )

        elif choice == "5":
            page = get_number("Page number: ")
            limit = get_number("Limit (1-200): ", 1, 200)
            api.get_orders(page, limit)

        elif choice == "6":
            order_id = get_number("Order ID: ")
            api.get_order(order_id)

        elif choice == "7":
            service_id = get_number("Automatic product Service ID: ")
            quantity = get_number("Quantity (1-100): ", 1, 100)

            if confirm_real_order():
                status, result = api.create_order(
                    service_id,
                    quantity,
                )

                if status == 200 and result and result.get("success"):
                    api.get_order(result["order"]["order_id"])
            else:
                print("Order cancelled.")

        elif choice == "8":
            print(
                "\nCustomer must open @raindeliverybot "
                "and press Start before ordering."
            )

            service_id = get_number("Manual product Service ID: ")
            quantity = get_number("Quantity (1-100): ", 1, 100)
            customer_id = get_number(
                "Customer numeric Telegram ID: "
            )

            if confirm_real_order():
                status, result = api.create_order(
                    service_id,
                    quantity,
                    delivery_telegram_id=customer_id,
                )

                if status == 200 and result and result.get("success"):
                    print(
                        "\nManual delivery will be sent "
                        "through @raindeliverybot."
                    )
                    api.get_order(result["order"]["order_id"])
            else:
                print("Order cancelled.")

        elif choice == "0":
            break

        else:
            print("Invalid option.")


if __name__ == "__main__":
    main()
